💡
How the system hangs together:
Raw sales data lands in GCS, gets cleaned by an Apache Beam job running on Dataflow, and flows into BigQuery. BigQuery ML's ARIMA_PLUS model trains directly inside the warehouse — no Python ML environment needed for the forecasting itself — and outputs 30-day predictions with confidence intervals per product.

Those predictions, combined with your business documents (past campaign reports, product notes), get embedded into FAISS or Chroma via LangChain. When a narrative is needed, the RAG chain retrieves the most relevant context for that product, and GPT-4 writes a 3-paragraph executive summary grounded in both the numbers and the business history.

Everything lands in PostgreSQL for low-latency serving via a FastAPI endpoint, with a Cloud Scheduler job re-running the full pipeline every week automatically.

🪣 Data Ingestion

  • Raw CSV sales files uploaded to GCS bucket
  • Apache Beam pipeline reads, validates, transforms
  • Runs on fully managed Dataflow service
  • Outputs clean Parquet to BigQuery staging table

📈 Forecasting

  • BQML ARIMA_PLUS trains inside BigQuery — pure SQL
  • 30-day horizon, per-product per-store granularity
  • Returns forecast + lower/upper confidence intervals
  • No external Python ML infra needed for training

🔗 RAG + Narrative

  • Business docs chunked and embedded with OpenAI text-embedding-3-small
  • FAISS (local) or Chroma (persistent) vector store
  • LangChain retrieval chain fetches top-K context
  • GPT-4 writes 3-paragraph executive summary per product

🚀 Serving

  • FastAPI writes forecasts + narratives to PostgreSQL
  • Two endpoints: /forecasts/{product} and /narrative/{product}
  • Cloud Scheduler triggers full pipeline weekly via HTTP
  • Async SQLAlchemy for non-blocking DB writes

Define the Beam pipeline
etl/beam_pipeline.py
›
Read CSVs from GCS, parse each row with a DoFn, validate and clean fields, then write to BigQuery.
python 
import apache_beam as beam
from apache_beam.options.pipeline_options import PipelineOptions
from apache_beam.io.gcp.bigquery import WriteToBigQuery, BigQueryDisposition
import csv, logging
from datetime import datetime

BQ_SCHEMA = {
    "fields": [
        {"name": "date",       "type": "DATE"},
        {"name": "product_id", "type": "STRING"},
        {"name": "store_id",   "type": "STRING"},
        {"name": "units_sold", "type": "FLOAT64"},
        {"name": "revenue",    "type": "FLOAT64"},
    ]
}

class ParseSalesRow(beam.DoFn):
    def process(self, line: str):
        try:
            row = next(csv.DictReader([line],
                fieldnames=["date","product_id","store_id","units_sold","revenue"]))
            # Validate and coerce
            yield {
                "date":       datetime.strptime(row["date"], "%Y-%m-%d").date().isoformat(),
                "product_id": row["product_id"].strip().upper(),
                "store_id":   row["store_id"].strip(),
                "units_sold": float(row["units_sold"] or 0),
                "revenue":    float(row["revenue"] or 0),
            }
        except (ValueError, KeyError) as e:
            logging.warning("Skipping bad row: %s — %s", line[:80], e)


def run(project: str, bucket: str, dataset: str, runner: str = "DirectRunner"):
    opts = PipelineOptions(
        runner      = runner,
        project     = project,
        region      = "us-central1",
        temp_location = f"gs://{bucket}/tmp",
        staging_location = f"gs://{bucket}/staging",
        save_main_session = True,
    )
    with beam.Pipeline(options=opts) as p:
        (
            p
            | "ReadCSV"      >> beam.io.ReadFromText(f"gs://{bucket}/sales/*.csv", skip_header_lines=1)
            | "ParseRows"    >> beam.ParDo(ParseSalesRow())
            | "WriteToBQ"    >> WriteToBigQuery(
                table          = f"{project}:{dataset}.sales_clean",
                schema         = BQ_SCHEMA,
                create_disposition = BigQueryDisposition.CREATE_IF_NEEDED,
                write_disposition  = BigQueryDisposition.WRITE_TRUNCATE,
            )
        )


if __name__ == "__main__":
    run(
        project = "my-gcp-project",
        bucket  = "my-retail-bucket",
        dataset = "retail_forecasting",
        runner  = "DataflowRunner",   # swap to DataflowRunner for prod
    )

Start here. BigQuery ML lets you train a production-grade time-series model with pure SQL — no Python ML environment, no GPU, no cluster management. Get a working 30-day forecast first, then build everything else around it.
1
Create the training dataset
BigQuery SQL
›
Aggregate daily sales per product into the time-series format that BQML ARIMA_PLUS expects: one row per (date, product_id) pair.
sql 
-- Create aggregated daily sales view
CREATE OR REPLACE VIEW `retail_forecasting.v_daily_sales` AS
SELECT
  date,
  product_id,
  SUM(units_sold) AS total_units,
  SUM(revenue)    AS total_revenue
FROM `retail_forecasting.sales_clean`
WHERE units_sold > 0           -- exclude zero-sale days from model
  AND date >= '2022-01-01'    -- at least 2 years of history
GROUP BY 1, 2
ORDER BY 1, 2;
2
Train ARIMA_PLUS model
BigQuery SQL
›
BQML automatically selects the best ARIMA order, handles holiday effects, and trains a separate model per product in one statement.
sql 
CREATE OR REPLACE MODEL `retail_forecasting.arima_sales_model`
OPTIONS(
  model_type          = 'ARIMA_PLUS',
  time_series_timestamp_col = 'date',
  time_series_data_col      = 'total_units',
  time_series_id_col        = 'product_id',   -- separate model per product
  horizon                   = 30,              -- 30-day forecast
  auto_arima                = TRUE,
  data_frequency            = 'AUTO_FREQUENCY',
  decompose_time_series     = TRUE,
  holiday_region            = 'US'             -- include US holiday effects
) AS
SELECT date, product_id, total_units
FROM `retail_forecasting.v_daily_sales`;
3
Generate 30-day forecasts
BigQuery SQL
›
Call ML.FORECAST to get per-product predictions with 80% and 95% confidence intervals, then materialize into a results table.
sql 
-- Generate forecast and store results
CREATE OR REPLACE TABLE `retail_forecasting.forecasts_30d` AS
SELECT
  forecast_timestamp                    AS forecast_date,
  product_id,
  ROUND(forecast_value, 2)             AS forecast_units,
  ROUND(prediction_interval_lower_bound, 2) AS ci_lower,
  ROUND(prediction_interval_upper_bound, 2) AS ci_upper,
  confidence_level,
  CURRENT_TIMESTAMP()                   AS generated_at
FROM
  ML.FORECAST(
    MODEL `retail_forecasting.arima_sales_model`,
    STRUCT(30 AS horizon, 0.8 AS confidence_level)
  )
ORDER BY product_id, forecast_date;
4
Evaluate model accuracy
BigQuery SQL
›
Use ML.EVALUATE to check MAPE, RMSE, and other metrics before promoting the model to production.
sql 
SELECT
  product_id,
  ROUND(mean_absolute_percentage_error * 100, 2) AS mape_pct,
  ROUND(root_mean_squared_error, 2)              AS rmse,
  ROUND(mean_absolute_error, 2)                  AS mae,
  symmetric_mean_absolute_percentage_error        AS smape,
  AIC
FROM
  ML.EVALUATE(MODEL `retail_forecasting.arima_sales_model`)
ORDER BY mape_pct ASC;

-- Check time-series components
SELECT *
FROM ML.ARIMA_EVALUATE(MODEL `retail_forecasting.arima_sales_model`)
ORDER BY AIC;

Build the vector store
rag/vector_store.py
›
Load and chunk all business documents, embed them with OpenAI's embedding model, and save the index for reuse.
python 
from pathlib import Path
from langchain_community.document_loaders import DirectoryLoader, PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import FAISS
import os

EMBED_MODEL = "text-embedding-3-small"
INDEX_PATH  = Path("data/faiss_index")

def build_vector_store(docs_dir: str = "data/business_docs") -> FAISS:
    # 1. Load PDFs and text files from the docs directory
    loader = DirectoryLoader(
        docs_dir,
        glob         = "**/*.{pdf,txt,md}",
        loader_cls   = PyPDFLoader,
        show_progress=True,
    )
    raw_docs = loader.load()

    # 2. Chunk into 600-token windows with 80-token overlap
    splitter = RecursiveCharacterTextSplitter(
        chunk_size    = 600,
        chunk_overlap = 80,
        separators    = ["\n\n", "\n", " ", ""],
    )
    chunks = splitter.split_documents(raw_docs)
    print(f"Loaded {len(raw_docs)} docs → {len(chunks)} chunks")

    # 3. Embed and store in FAISS
    embeddings = OpenAIEmbeddings(model=EMBED_MODEL)
    store      = FAISS.from_documents(chunks, embeddings)

    # 4. Persist index to disk
    INDEX_PATH.mkdir(parents=True, exist_ok=True)
    store.save_local(str(INDEX_PATH))
    return store


def load_vector_store() -> FAISS:
    embeddings = OpenAIEmbeddings(model=EMBED_MODEL)
    return FAISS.load_local(
        str(INDEX_PATH), embeddings,
        allow_dangerous_deserialization=True,
    )


if __name__ == "__main__":
    build_vector_store()
    print("Vector store saved to", INDEX_PATH)

Build the RAG narrative chain
rag/narrative_chain.py
›
Combine the retrieved context with the numeric forecast data and pass both to GPT-4 to generate a grounded 3-paragraph executive summary.
python 
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.runnable import RunnablePassthrough
from langchain.schema.output_parser import StrOutputParser
from rag.vector_store import load_vector_store

_SYSTEM = """You are a senior retail analyst. Write a concise 3-paragraph executive
summary for the following product forecast. Ground every claim in either the
numeric forecast data or the retrieved business context — do not invent facts.

Paragraph 1 — Trend: describe the 30-day sales trajectory.
Paragraph 2 — Drivers: explain likely causes using the business context.
Paragraph 3 — Recommendation: one concrete action the merchandising team should take."""

_HUMAN = """PRODUCT: {product_id}

FORECAST DATA:
{forecast_data}

RETRIEVED BUSINESS CONTEXT:
{context}

Write the executive summary now."""

def build_chain():
    store     = load_vector_store()
    retriever = store.as_retriever(search_kwargs={"k": 5})
    llm       = ChatOpenAI(model="gpt-4o", temperature=0.3)
    prompt    = ChatPromptTemplate.from_messages([
        ("system", _SYSTEM),
        ("human",  _HUMAN),
    ])

    def format_docs(docs) -> str:
        return "\n\n---\n\n".join(
            f"[{d.metadata.get('source','?')}]\n{d.page_content}" for d in docs
        )

    return (
        {
            "context":       retriever | format_docs,
            "product_id":    RunnablePassthrough(),
            "forecast_data": RunnablePassthrough(),
        }
        | prompt
        | llm
        | StrOutputParser()
    )


def generate_narrative(product_id: str, forecast_rows: list[dict]) -> str:
    chain = build_chain()
    forecast_str = "\n".join(
        f"{r['forecast_date']}: {r['forecast_units']:.0f} units "
        f"(CI: {r['ci_lower']:.0f}–{r['ci_upper']:.0f})"
        for r in forecast_rows
    )
    return chain.invoke({
        "product_id":    product_id,
        "forecast_data": forecast_str,
    })


if __name__ == "__main__":
    import json
    sample = [
        {"forecast_date":"2024-02-01","forecast_units":145,"ci_lower":130,"ci_upper":160},
    ]
    narrative = generate_narrative("SKU-001", sample)
    print(narrative)

Database schema & models
db/models.py
›
Define SQLAlchemy ORM models for the forecasts and narratives tables in PostgreSQL.
python 
from sqlalchemy import Column, String, Float, Date, DateTime, Text, Integer
from sqlalchemy.orm import DeclarativeBase
from datetime import datetime, timezone


class Base(DeclarativeBase):
    pass


class Forecast(Base):
    __tablename__ = "forecasts"

    id            = Column(Integer, primary_key=True, autoincrement=True)
    product_id    = Column(String(64),  nullable=False, index=True)
    forecast_date = Column(Date,         nullable=False)
    forecast_units= Column(Float,        nullable=False)
    ci_lower      = Column(Float)
    ci_upper      = Column(Float)
    generated_at  = Column(DateTime(timezone=True),
                           default=lambda: datetime.now(timezone.utc))


class Narrative(Base):
    __tablename__ = "narratives"

    id          = Column(Integer, primary_key=True, autoincrement=True)
    product_id  = Column(String(64), nullable=False, index=True)
    summary     = Column(Text,       nullable=False)
    generated_at= Column(DateTime(timezone=True),
                         default=lambda: datetime.now(timezone.utc))

FastAPI application
api/main.py
›
Expose /forecasts/{product_id}, /narrative/{product_id}, and the /run-pipeline trigger endpoint. Uses async SQLAlchemy for non-blocking DB access.
python 
from contextlib import asynccontextmanager
from fastapi import FastAPI, HTTPException, BackgroundTasks, Depends, Header
from fastapi.responses import JSONResponse
from sqlalchemy.ext.asyncio import AsyncSession, create_async_engine, async_sessionmaker
from sqlalchemy import select
from pydantic import BaseModel
from typing import Annotated
import os, secrets
from db.models import Base, Forecast, Narrative

DATABASE_URL = os.environ["DATABASE_URL"]   # e.g. postgresql+asyncpg://...
SCHEDULER_SECRET = os.environ["SCHEDULER_SECRET"]  # shared secret for /run-pipeline

engine     = create_async_engine(DATABASE_URL, pool_size=10)
SessionDep = async_sessionmaker(engine, expire_on_commit=False)


async def get_db():
    async with SessionDep() as db:
        yield db


@asynccontextmanager
async def lifespan(app):
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    yield


app = FastAPI(title="Retail Forecasting API", version="1.0", lifespan=lifespan)


class ForecastRow(BaseModel):
    forecast_date:  str
    forecast_units: float
    ci_lower:       float | None
    ci_upper:       float | None


@app.get("/forecasts/{product_id}", response_model=list[ForecastRow])
async def get_forecasts(product_id: str, db: AsyncSession = Depends(get_db)):
    result = await db.execute(
        select(Forecast)
        .where(Forecast.product_id == product_id.upper())
        .order_by(Forecast.forecast_date)
    )
    rows = result.scalars().all()
    if not rows:
        raise HTTPException(404, f"No forecasts for product {product_id!r}")
    return [ForecastRow(
        forecast_date  = str(r.forecast_date),
        forecast_units = r.forecast_units,
        ci_lower       = r.ci_lower,
        ci_upper       = r.ci_upper,
    ) for r in rows]


@app.get("/narrative/{product_id}")
async def get_narrative(product_id: str, db: AsyncSession = Depends(get_db)):
    result = await db.execute(
        select(Narrative)
        .where(Narrative.product_id == product_id.upper())
        .order_by(Narrative.generated_at.desc())
        .limit(1)
    )
    row = result.scalar_one_or_none()
    if not row:
        raise HTTPException(404, f"No narrative for product {product_id!r}")
    return {"product_id": row.product_id, "summary": row.summary, "generated_at": str(row.generated_at)}


@app.post("/run-pipeline", status_code=202)
async def trigger_pipeline(
    background_tasks: BackgroundTasks,
    x_scheduler_secret: Annotated[str | None, Header()] = None,
):
    # Validate shared secret from Cloud Scheduler OIDC header
    if not secrets.compare_digest(x_scheduler_secret or "", SCHEDULER_SECRET):
        raise HTTPException(403, "Forbidden")
    background_tasks.add_task(run_full_pipeline)
    return {"status": "pipeline queued"}


async def run_full_pipeline():
    # Orchestrate: Beam ETL → BQML retrain → forecast → RAG narrative → DB write
    import subprocess, asyncio
    await asyncio.to_thread(subprocess.run, ["python", "-m", "etl.beam_pipeline"], check=True)
    # BQML retrain and forecast run via BigQuery job API (see bq_jobs.py)
